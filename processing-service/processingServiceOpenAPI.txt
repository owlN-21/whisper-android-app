openapi: 3.0.3
info:
  title: Processing Service API
  version: 1.0.0
  description: API for two-step processing audio to transcript, transcript to summary

servers:
  - url: http://localhost:8090
    description: Local processing-service

tags:
  - name: Transcriptions
    description: Audio transcription operations
  - name: Summaries
    description: Transcript summarization operations

paths:
  /api/v1/transcriptions:
    post:
      tags:
        - Transcriptions
      summary: Submit audio for transcription
      operationId: createTranscription
      requestBody:
        required: true
        content:
          multipart/form-data:
            schema:
              type: object
              required:
                - taskId
                - file
              properties:
                taskId:
                  type: integer
                  format: int64
                  example: 15
                file:
                  type: string
                  format: binary
      responses:
        '202':
          description: Transcription task accepted
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ProcessingAcceptedResponse'
        '400':
          description: Invalid transcription request
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '409':
          description: Task already exists
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '500':
          description: Internal processing error
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'

  /api/v1/transcriptions/{taskId}:
    get:
      tags:
        - Transcriptions
      summary: Get transcription result by task id
      operationId: getTranscriptionByTaskId
      parameters:
        - name: taskId
          in: path
          required: true
          schema:
            type: integer
            format: int64
          example: 15
      responses:
        '200':
          description: Transcription status or result returned
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/TranscriptionResultResponse'
        '404':
          description: Task not found
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '500':
          description: Internal processing error
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'

  /api/v1/summaries:
    post:
      tags:
        - Summaries
      summary: Submit transcript for summarization
      operationId: createSummary
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required:
                - taskId
                - text
              properties:
                taskId:
                  type: integer
                  format: int64
                  example: 15
                text:
                  type: string
                  example: Это распознанный текст лекции.
      responses:
        '202':
          description: Summary task accepted
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ProcessingAcceptedResponse'
        '400':
          description: Invalid summary request
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '409':
          description: Task already exists
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '500':
          description: Internal summarization error
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'

  /api/v1/summaries/{taskId}:
    get:
      tags:
        - Summaries
      summary: Get summary result by task id
      operationId: getSummaryByTaskId
      parameters:
        - name: taskId
          in: path
          required: true
          schema:
            type: integer
            format: int64
          example: 15
      responses:
        '200':
          description: Summary status or result returned
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/SummaryResultResponse'
        '404':
          description: Task not found
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'
        '500':
          description: Internal summarization error
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/ApiErrorResponse'

components:
  schemas:
    ProcessingAcceptedResponse:
      type: object
      required:
        - taskId
        - status
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - ACCEPTED
          example: ACCEPTED

    TranscriptionResultResponse:
      oneOf:
        - $ref: '#/components/schemas/TranscriptionInProgressResponse'
        - $ref: '#/components/schemas/TranscriptionCompletedResponse'
        - $ref: '#/components/schemas/TranscriptionFailedResponse'

    TranscriptionInProgressResponse:
      type: object
      required:
        - taskId
        - status
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - IN_PROGRESS
          example: IN_PROGRESS

    TranscriptionCompletedResponse:
      type: object
      required:
        - taskId
        - status
        - text
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - COMPLETED
          example: COMPLETED
        text:
          type: string
          example: Это распознанный текст лекции.

    TranscriptionFailedResponse:
      type: object
      required:
        - taskId
        - status
        - errorMessage
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - FAILED
          example: FAILED
        errorMessage:
          type: string
          example: ASR model unavailable

    SummaryResultResponse:
      oneOf:
        - $ref: '#/components/schemas/SummaryInProgressResponse'
        - $ref: '#/components/schemas/SummaryCompletedResponse'
        - $ref: '#/components/schemas/SummaryFailedResponse'

    SummaryInProgressResponse:
      type: object
      required:
        - taskId
        - status
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - IN_PROGRESS
          example: IN_PROGRESS

    SummaryCompletedResponse:
      type: object
      required:
        - taskId
        - status
        - content
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - COMPLETED
          example: COMPLETED
        content:
          type: string
          example: |
            1. Основная тема лекции
            2. Ключевые понятия
            3. Итоговые выводы

    SummaryFailedResponse:
      type: object
      required:
        - taskId
        - status
        - errorMessage
      properties:
        taskId:
          type: integer
          format: int64
          example: 15
        status:
          type: string
          enum:
            - FAILED
          example: FAILED
        errorMessage:
          type: string
          example: Summarization model unavailable

    ApiErrorResponse:
      type: object
      required:
        - description
        - code
      properties:
        description:
          type: string
          example: Task not found
        code:
          type: string
          example: TASK_NOT_FOUND
        exceptionName:
          type: string
          nullable: true
          example: TaskNotFoundException
        exceptionMessage:
          type: string
          nullable: true
          example: Task with id 15 was not found
        stacktrace:
          type: array
          nullable: true
          items:
            type: string
